١٢

ذخائر لبنان

في بعض الأماكن من تلك السهول: مما غيَّر بعض الشيء في وضعها. فرأينا أن نلخص منها ما يتعلق بمطابها. وإنا لنأسف كما أَسفَ حضرته على ما فات المؤرخين في أوصافهم المسهبة من جراء ما زاد على أعمال مطلبنا، ولم نعثر على ما يتأكد به صحة ذلك صادرًا عن أبناء تلك المعقِّبة الخاصة سواد؛ فلا نبرتهم من العيب التأريخي بما هو ظاهر على كلامهم من صحة التحزب.

إنَّ ما يصيب النُّفُور الشَّاملة من الزلازل جار بين خطين متوازيين على قدر معلوم، ثم ينحرفان إلى ملتقى واحد عند حلب بشكل زاوية حادة؛ فالخط الأول - وهو الغربي - يبدأ عند مجرى دجلة السفلي بقرب ديار بكر ويجري إلى الرها «أورفه»، فينحرف فحلٌ فأنطاكيَّة، ثم يميل إلى الجنوب فيمر بساحل البحر إلى عسقلان وغزة حيث ينتهي، ويماء عليه فإنَّ هذا الخط يجتاز جميع ساحل لبنان، والخط الثاني - وهو الشرقي - يبدأ عند عينتاب، وينحدر مستقيمًا نحو الجنوب، ويقطع الخط الأول عند حلب، ثم يجوز سائرًا في وسط وادي العاصي ووهاد بلاد البقاع إلى غور الأردن.

ففي مواقع الخط الأول من البلاد الجاورة للبحر المتوسط وقع على ما رواه أصحاب الآثار عدة من الزلازل في السنين الآتية للمسيح: وهي: سنة ١٣١، و٢٦٠، و٣٣٣، و ٣٤٤، و٣٨٧، و٤٤٤، و٤٥٨، و٤٩٢، و٥٢٩، و٥٦٠، و٥٨٠، و٥٨٩، و٧١٣، و٧٧٥، و٨٥٣، و١٠١٦، و١٠٢٢، و١٠٣٣، و١٠٦٢، و١٠٦٩، و١١٠٩، و١١٢٩، و١١٥٥، و١٢٠٤، و١٢١٢، و١٢٢٩، و١٣٣٩، و١٤٠٢، و١٥٤٦، و١٦٥٥، و١٧٨٣، و١٧٩٦، و١٨٢٢، و١٨٥٩، و١٨٧٢، و١٨٧٣.

وحدثت عدة زلازل قبل المسيح منها ما أخبر به إسترابون في عرض الكلام على الموقعة التي جرت في سنة ١٤٣ قبل المسيح بين أهل عكه والقائد السوريين؛ إذ قال: «جاشت مياه البحر بين عكه وصور، وامتدت كما في الد، وأغرقت مَنْ فرَّ هاربًا من جنود سوريدون، ولما خُبِّرت المياهُ وُجد جيشهم على سيف البحر بين الأسماك الهالكة».

ومن أعظم الزلازل الزائلة التي حدثت أيام يوستينيانوس الأول في سنة ٥٢٥، واشتهر ذكرها في التاريخ، وبقيت آثارها إلى أيامنا هذه دالَّةً على ما حدث في بعض الأماكن من تغيير هيئة الساحل. قال المُؤرِّخ ثاوفان: «إنَّ رأس الشقعة موضع بين البترون وطرابلس، قد تدفَّق يومئذٍ إلى البحر، وصار في مكانه خور واسع، وأأصّح
